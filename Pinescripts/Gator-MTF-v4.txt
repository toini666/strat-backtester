// ============================================
// NOTES D'ARCHITECTURE — V4
// ============================================
// - CHART M1, TRIGGER M7 (request.security, lookahead_on + [1] sur chaque sortie =
//   idiome canonique Pine v6 non-repaint, garantit qu'on lit toujours la DERNIÈRE
//   bougie HTF CONFIRMÉE — identique en mode historique et en replay/real-time
//   (lookahead_off seul est asymétrique : confirmé en historique, mais bougie en
//    cours de construction en real-time, ce qui faisait que ta.cross fire trop tôt
//    dans la M7 en replay).
//
// CONVENTION COULEUR CANAL HMA (chart + trigger) — stratégie MEAN-REVERSION :
//     canalVert  = HMA1 < HMA2 → canal DESCENDANT → on cherche un LONG (reversal après baisse) → affiché en VERT
//     canalRouge = HMA1 > HMA2 → canal ASCENDANT  → on cherche un SHORT (reversal après hausse) → affiché en ROUGE
//   (la couleur reflète l'INTENTION de trade, pas la direction du canal :
//    vert = on est en recherche de long, rouge = on est en recherche de short.
//    Conséquence directe : un cross HMA au-dessus du SSL upper sur canal rouge
//    (HMA ascendant) ouvre une fenêtre SHORT (fade du breakout haussier),
//    et un cross HMA sous le SSL lower sur canal vert ouvre une fenêtre LONG.)
//
// - Entrées : fenêtre ouverte par cross BIDIRECTIONNEL HMA M7 vs SSL upper/lower, gaté par la couleur du canal.
//     LONG signal  : ta.cross(HMA, SSL upper) ET canal vert (HMA descendant)   → ouvre windowLong
//                    (cas typique : HMA redescend sous SSL upper après un sommet, début de descent)
//     SHORT signal : ta.cross(HMA, SSL lower) ET canal rouge (HMA ascendant)   → ouvre windowShort
//                    (cas typique : HMA remonte au-dessus de SSL lower après un creux, début de remontée)
//   La fenêtre est exprimée en BOUGIES TRIGGER (par défaut 5 → 35 bougies M1 en chart M1).
// - 3 conditions binaires :
//     C1 = couleur canal HMA M7 (canalRouge = asc / canalVert = desc)
//     C2 = couleur canal HMA M1 (idem)
//     C3 = couleur MFI Smart Money Flow M1 (oscBull si mfiM1>0, oscBear si mfiM1<0)
//   Pondération unique = sig hyperwave 4Kings en extrême (un seul seuil, défaut 40).
//   Long  : sig en extrême OK si oscSig ≤ -seuil.
//   Short : sig en extrême OK si oscSig ≥ +seuil.
// - 4 cas activables pour LONG (sur windowLong — miroir pour SHORT sur windowShort) :
//     A : C1 vert  + C2 vert  + C3 bull → entrée directe
//     B : C1 vert  + C2 vert  + C3 bear → entrée directe
//     C : C1 rouge + C2 vert  + C3 bear → marqueur orange, entrée si sig extrême OK
//     D : C1 rouge + C2 vert  + C3 bull → marqueur orange, entrée si sig extrême OK
//   Cas E informatif : C1 vert + C2 rouge + C3 bear → marqueur gris, aucune entrée.
//   (SHORT miroir : windowShort, swap vert↔rouge et bull↔bear)
// - Sorties : SL initial (lookback + min %), TP partiel à RR i_partialRr (puis SL à BE),
//   TP final à RR i_finalRr (sortie totale), auto-close 21h Bruxelles, blackouts.
//   Plus aucune sortie sur cross HMA opposé ni sur HW.
// - "Un trade max par fenêtre" : suivi via le bar M1 de détection du cross M7.

roundToTick(float price) =>
    math.round(price / syminfo.mintick) * syminfo.mintick

formatTimeframeLabel(string rawTf) =>
    float tfNum = str.tonumber(rawTf)
    string result = if not na(tfNum)
        int tfMinutes = int(tfNum)
        tfMinutes >= 60 and tfMinutes % 60 == 0 ? "H" + str.tostring(tfMinutes / 60) : "M" + str.tostring(tfMinutes)
    else if str.endswith(rawTf, "S")
        "S" + str.substring(rawTf, 0, str.length(rawTf) - 1)
    else if rawTf == "D"
        "D1"
    else if str.endswith(rawTf, "D")
        "D" + str.substring(rawTf, 0, str.length(rawTf) - 1)
    else if rawTf == "W"
        "W1"
    else if str.endswith(rawTf, "W")
        "W" + str.substring(rawTf, 0, str.length(rawTf) - 1)
    else if rawTf == "M"
        "MO1"
    else if str.endswith(rawTf, "M")
        "MO" + str.substring(rawTf, 0, str.length(rawTf) - 1)
    else
        rawTf
    result

getTimeframe() => formatTimeframeLabel(timeframe.period)

isInTimeSlot(int hh, int mm, int startH, int startM, int endH, int endM) =>
    int curMin   = hh * 60 + mm
    int startMin = startH * 60 + startM
    int endMin   = endH * 60 + endM
    bool endIsDayLastMinute = endH == 23 and endM == 59
    if startMin <= endMin
        curMin >= startMin and (endIsDayLastMinute ? curMin <= endMin : curMin < endMin)
    else
        curMin >= startMin or (endIsDayLastMinute ? curMin <= endMin : curMin < endMin)

isBlackoutWindow() =>
    int hh = hour(time_close, "Europe/Brussels")
    int mm = minute(time_close, "Europe/Brussels")
    bool b1 = i_bo1On and isInTimeSlot(hh, mm, i_bo1StartH, i_bo1StartM, i_bo1EndH, i_bo1EndM)
    bool b2 = i_bo2On and isInTimeSlot(hh, mm, i_bo2StartH, i_bo2StartM, i_bo2EndH, i_bo2EndM)
    bool b3 = i_bo3On and isInTimeSlot(hh, mm, i_bo3StartH, i_bo3StartM, i_bo3EndH, i_bo3EndM)
    bool b4 = i_bo4On and isInTimeSlot(hh, mm, i_bo4StartH, i_bo4StartM, i_bo4EndH, i_bo4EndM)
    bool b5 = i_bo5On and isInTimeSlot(hh, mm, i_bo5StartH, i_bo5StartM, i_bo5EndH, i_bo5EndM)
    bool b6 = i_bo6On and isInTimeSlot(hh, mm, i_bo6StartH, i_bo6StartM, i_bo6EndH, i_bo6EndM)
    b1 or b2 or b3 or b4 or b5 or b6

triggerCalc() =>
    float srcEma = ta.ema(close, i_emaLen)
    float rawH1  = ta.hma(srcEma, i_hma1Len)
    float rawH2  = ta.hma(srcEma, i_hma2Len)
    float h1     = srcEma + (rawH1 - srcEma) * i_ampMult
    float h2     = srcEma + (rawH2 - srcEma) * i_ampMult

    float BBMC    = ta.wma(2 * ta.wma(close, len_ssl / 2) - ta.wma(close, len_ssl), math.round(math.sqrt(len_ssl)))
    float rangema = ta.ema(ta.tr, len_ssl)
    float upperk  = BBMC + rangema * multy_ssl
    float lowerk  = BBMC - rangema * multy_ssl

    // Couleur canal au bar courant (utilisée pour gater les signaux + remontée vers le main scope).
    bool canalRouge_t = h1 > h2   // ascendant
    bool canalVert_t  = h1 < h2   // descendant

    // Signaux de cross — bidirectionnels (ta.cross), gatés par la couleur du canal :
    //   LONG signal  = HMA croise (haut ou bas) le SSL upper PENDANT canal vert (HMA descendant)
    //                  → typique : HMA repasse sous SSL upper après un sommet, début de descent.
    //   SHORT signal = HMA croise (haut ou bas) le SSL lower PENDANT canal rouge (HMA ascendant)
    //                  → typique : HMA repasse au-dessus de SSL lower après un creux, début de remontée.
    bool fastLongSig  = ta.cross(h1, upperk) and canalVert_t
    bool fastShortSig = ta.cross(h1, lowerk) and canalRouge_t
    bool slowLongSig  = ta.cross(h2, upperk) and canalVert_t
    bool slowShortSig = ta.cross(h2, lowerk) and canalRouge_t

    // [1] sur CHAQUE sortie + barmerge.lookahead_on côté request.security = idiome
    // canonique TradingView pour "valeurs HTF confirmées only, identique historique
    // et real-time". Sans ça, en replay, ta.cross fire dès qu'il se produit dans la
    // M7 en cours, au lieu d'attendre la clôture confirmée comme en historique.
    [h1[1], h2[1], BBMC[1], upperk[1], lowerk[1], fastLongSig[1], fastShortSig[1], slowLongSig[1], slowShortSig[1], canalRouge_t[1], time[1]]

[hma1T, hma2T, sslBaseT, sslUpT, sslLoT, fastLongSig_t, fastShortSig_t, slowLongSig_t, slowShortSig_t, canalRougeRaw, timeT] = request.security(syminfo.tickerid, i_triggerTf, triggerCalc(), barmerge.gaps_off, barmerge.lookahead_on)

bool slowCrossLong  = newTriggerBar and (slowLongSig_t  == true)
bool slowCrossShort = newTriggerBar and (slowShortSig_t == true)

// Couleur canal HMA trigger — convention mean-reversion (cf. notes d'architecture)
bool canalRougeT = canalRougeRaw == true   // canal ascendant → recherche SHORT → ROUGE
bool canalVertT  = canalRougeRaw == false  // canal descendant → recherche LONG → VERT

if fastCrossLong
    lastFastLongCrossBar := bar_index
if fastCrossShort
    lastFastShortCrossBar := bar_index
if slowCrossLong
    lastSlowLongCrossBar := bar_index
if slowCrossShort
    lastSlowShortCrossBar := bar_index

// ============================================
// FENÊTRES D'ENTRÉE
// ============================================

bool windowFastLong  = not na(lastFastLongCrossBar)  and (bar_index - lastFastLongCrossBar)  <= windowBarsChart
bool windowFastShort = not na(lastFastShortCrossBar) and (bar_index - lastFastShortCrossBar) <= windowBarsChart
bool windowSlowLong  = not na(lastSlowLongCrossBar)  and (bar_index - lastSlowLongCrossBar)  <= windowBarsChart
bool windowSlowShort = not na(lastSlowShortCrossBar) and (bar_index - lastSlowShortCrossBar) <= windowBarsChart

bool windowLong  = windowFastLong  or windowSlowLong
bool windowShort = windowFastShort or windowSlowShort

// ============================================
// CONDITIONS D'ENTRÉE — CAS A/B/C/D + CAS E (info)
// ============================================
// Pour un long : "ok" = bull. Pour un short : "ok" = bear (miroir).

// --- Setups armés (sans la condition sig extrême pour C/D) ---
// LONG (mean-reversion : windowLong ouvert par cross HMA/SSL upper PENDANT canal vert)
bool armedA_Long = windowLong and canalVertT  and canalVertM1  and oscBullishM1
bool armedB_Long = windowLong and canalVertT  and canalVertM1  and oscBearishM1
bool armedC_Long = windowLong and canalRougeT and canalVertM1  and oscBearishM1
bool armedD_Long = windowLong and canalRougeT and canalVertM1  and oscBullishM1
bool armedE_Long = windowLong and canalVertT  and canalRougeM1 and oscBearishM1
// SHORT (miroir : windowShort ouvert par cross HMA/SSL lower PENDANT canal rouge)
bool armedA_Short = windowShort and canalRougeT and canalRougeM1 and oscBearishM1
bool armedB_Short = windowShort and canalRougeT and canalRougeM1 and oscBullishM1
bool armedC_Short = windowShort and canalVertT  and canalRougeM1 and oscBullishM1
bool armedD_Short = windowShort and canalVertT  and canalRougeM1 and oscBearishM1
bool armedE_Short = windowShort and canalRougeT and canalVertM1  and oscBullishM1

// --- Entrées effectives ---
bool entryA_Long = i_caseA_On and armedA_Long
bool entryB_Long = i_caseB_On and armedB_Long
bool entryC_Long = i_caseC_On and armedC_Long and sigExtremeBearM1
bool entryD_Long = i_caseD_On and armedD_Long and sigExtremeBearM1
bool entryA_Short = i_caseA_On and armedA_Short
bool entryB_Short = i_caseB_On and armedB_Short
bool entryC_Short = i_caseC_On and armedC_Short and sigExtremeBullM1
bool entryD_Short = i_caseD_On and armedD_Short and sigExtremeBullM1

bool longSignalRaw  = entryA_Long  or entryB_Long  or entryC_Long  or entryD_Long
bool shortSignalRaw = entryA_Short or entryB_Short or entryC_Short or entryD_Short

// Label du cas déclenché (priorité A>B>C>D)
string firedCaseRaw = entryA_Long or entryA_Short ? "A" : entryB_Long or entryB_Short ? "B" : entryC_Long or entryC_Short ? "C" : entryD_Long or entryD_Short ? "D" : ""

// Famille du cross (Fast prioritaire si les deux fenêtres sont actives au même bar — la grille est identique)
// Mean-reversion : LONG fire sur windowLong, SHORT sur windowShort → on bind l'identification à la bonne fenêtre.
bool useFastForLong  = windowFastLong
bool useFastForShort = windowFastShort

// "Un trade par fenêtre" : on bloque si on a déjà tradé la même fenêtre (cross trigger)
bool fastWindowOk = not i_oneTradePerWindow or na(lastFastTradedBar) or
   (lastFastLongCrossBar != lastFastTradedBar and lastFastShortCrossBar != lastFastTradedBar)
bool slowWindowOk = not i_oneTradePerWindow or na(lastSlowTradedBar) or
   (lastSlowLongCrossBar != lastSlowTradedBar and lastSlowShortCrossBar != lastSlowTradedBar)

bool longWindowOk  = useFastForLong  ? fastWindowOk : slowWindowOk
bool shortWindowOk = useFastForShort ? fastWindowOk : slowWindowOk

bool longSignal  = longSignalRaw  and longWindowOk
bool shortSignal = shortSignalRaw and shortWindowOk

// ============================================
// CALCUL SL & TPs (hybride high/low + min %)
// ============================================

float slBuffer = i_tickBuffer * syminfo.mintick

float rawSlLongBase  = ta.lowest(low,  i_slLookback) - slBuffer
float rawSlShortBase = ta.highest(high, i_slLookback) + slBuffer

float minSlDist       = close * (i_slMinPct / 100.0)
float minSlLongFloor  = close - minSlDist
float minSlShortFloor = close + minSlDist

float calcStopLong   = rawSlLongBase  > minSlLongFloor  ? minSlLongFloor  : rawSlLongBase
float calcStopShort  = rawSlShortBase < minSlShortFloor ? minSlShortFloor : rawSlShortBase

float calcEntryLong  = close
float calcEntryShort = close

bool slLongOk  = calcStopLong  < close
bool slShortOk = calcStopShort > close

bool cooldownOk = (bar_index - lastCloseBarIndex) >= i_cooldownBars

bool finalLongEntry  = longSignal  and slLongOk  and cooldownOk
bool finalShortEntry = shortSignal and slShortOk and cooldownOk

// ============================================
// ALERTES JSON
// ============================================

triggerAlert(string alertType, string alertSide, float alertEntry, float alertStop, float alertTp) =>
    string alertTf = getTimeframe()
    string json = '{\n'
    json += '  "ticker": "' + syminfo.ticker + '",\n'
    json += '  "type": "' + alertType + '",\n'
    json += '  "side": "' + alertSide + '",\n'
    if alertType == "PARTIAL" or alertType == "CLOSE"
        if not na(alertEntry)
            json += '  "entry": ' + str.tostring(alertEntry) + ',\n'
    else if alertType == "SIGNAL"
        if not na(alertEntry)
            json += '  "entry": ' + str.tostring(alertEntry) + ',\n'
        if not na(alertStop)
            json += '  "stop": ' + str.tostring(alertStop) + ',\n'
        if not na(alertTp)
            json += '  "tp": ' + str.tostring(alertTp) + ',\n'
    json += '  "strat": "' + i_stratName + '-' + syminfo.ticker + '-' + alertTf + '",\n'
    json += '  "timeframe": "' + alertTf + '"\n'
    json += '}'
    alert(json, alert.freq_once_per_bar_close)

// GESTION DES POSITIONS
// ============================================

bool canAct = bar_index > lastActionBar

// --- 0. AUTO-CLOSE FORCÉE ---
bool forceCloseTriggered = false
int  forceCloseSide      = 0

if positionActive and i_autoCloseOn and autoCloseBar and canAct
    forceCloseTriggered := true
    forceCloseSide      := currentSide
    positionActive      := false
    currentSide         := 0
    lastActionBar       := bar_index
    lastCloseBarIndex   := bar_index
    tpPartialHit        := false
    firedCaseLabel      := na
    canAct              := false
    if not na(lineEntry)
        line.delete(lineEntry)
        lineEntry := na
    if not na(lineStop)
        line.delete(lineStop)
        lineStop := na
    if not na(lineTpPartial)
        line.delete(lineTpPartial)
        lineTpPartial := na
    if not na(lineTpFinal)
        line.delete(lineTpFinal)
        lineTpFinal := na

// --- 1. ENTRÉE LONGUE ---
if not positionActive and finalLongEntry and canAct and not inBlackout
    positionActive   := true
    currentSide      := 1
    entryPrice       := roundToTick(calcEntryLong)
    stopPrice        := roundToTick(calcStopLong)
    initialStopPrice := stopPrice
    float riskLong   = entryPrice - initialStopPrice
    tpPartialPrice   := i_partialRr > 0 and i_hwPartialPct > 0 ? roundToTick(entryPrice + riskLong * i_partialRr) : na
    tpFinalPrice     := roundToTick(entryPrice + riskLong * i_finalRr)
    lastActionBar    := bar_index
    tpPartialHit     := false
    firedCaseLabel   := firedCaseRaw
    if useFastForLong
        lastFastTradedBar := lastFastLongCrossBar
    else
        lastSlowTradedBar := lastSlowLongCrossBar
    lineEntry     := line.new(bar_index, entryPrice, bar_index + 1, entryPrice, color=color.blue, style=line.style_dotted, width=1)
    lineStop      := line.new(bar_index, stopPrice,  bar_index + 1, stopPrice,  color=color.red,  style=line.style_dotted, width=1)
    if not na(tpPartialPrice)
        lineTpPartial := line.new(bar_index, tpPartialPrice, bar_index + 1, tpPartialPrice, color=color.green, style=line.style_dotted, width=1)
    lineTpFinal   := line.new(bar_index, tpFinalPrice, bar_index + 1, tpFinalPrice, color=color.green, style=line.style_dotted, width=1)
    triggerAlert("SIGNAL", "BUY", entryPrice, stopPrice, tpFinalPrice)

// --- 2. ENTRÉE COURTE ---
if not positionActive and finalShortEntry and canAct and not inBlackout
    positionActive   := true
    currentSide      := -1
    entryPrice       := roundToTick(calcEntryShort)
    stopPrice        := roundToTick(calcStopShort)
    initialStopPrice := stopPrice
    float riskShort  = initialStopPrice - entryPrice
    tpPartialPrice   := i_partialRr > 0 and i_hwPartialPct > 0 ? roundToTick(entryPrice - riskShort * i_partialRr) : na
    tpFinalPrice     := roundToTick(entryPrice - riskShort * i_finalRr)
    lastActionBar    := bar_index
    tpPartialHit     := false
    firedCaseLabel   := firedCaseRaw
    if useFastForShort
        lastFastTradedBar := lastFastShortCrossBar
    else
        lastSlowTradedBar := lastSlowShortCrossBar
    lineEntry     := line.new(bar_index, entryPrice, bar_index + 1, entryPrice, color=color.blue, style=line.style_dotted, width=1)
    lineStop      := line.new(bar_index, stopPrice,  bar_index + 1, stopPrice,  color=color.red,  style=line.style_dotted, width=1)
    if not na(tpPartialPrice)
        lineTpPartial := line.new(bar_index, tpPartialPrice, bar_index + 1, tpPartialPrice, color=color.green, style=line.style_dotted, width=1)
    lineTpFinal   := line.new(bar_index, tpFinalPrice, bar_index + 1, tpFinalPrice, color=color.green, style=line.style_dotted, width=1)
    triggerAlert("SIGNAL", "SELL", entryPrice, stopPrice, tpFinalPrice)

// --- 3. SORTIES (SL, TP partiel, BE, TP final) ---
bool hitStopLoss   = false
bool hitBreakeven  = false
bool hitFinalExit  = false
bool hitTpPartial  = false

if positionActive and canAct and bar_index > lastActionBar
    if currentSide == 1  // LONG
        // SL initial (pré-partiel) ou BE (post-partiel)
        if low <= stopPrice and not tpPartialHit[1]
            hitStopLoss := true
        if tpPartialHit and low <= stopPrice and not hitStopLoss
            hitBreakeven := true

        // TP final (sortie totale) — priorité sur le TP partiel pour gérer les configs RR_final ≤ RR_partial
        if not na(tpFinalPrice) and high >= tpFinalPrice and not hitStopLoss and not hitBreakeven
            hitFinalExit := true

        // TP partiel (uniquement si TP final pas touché)
        if not tpPartialHit and not na(tpPartialPrice) and high >= tpPartialPrice and not hitStopLoss and not hitFinalExit
            hitTpPartial := true
            tpPartialHit := true
            stopPrice    := entryPrice
            if not na(lineStop)
                line.delete(lineStop)
            lineStop := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.orange, style=line.style_dotted, width=1)
            if not na(lineTpPartial)
                line.delete(lineTpPartial)
                lineTpPartial := na
            triggerAlert("PARTIAL", "SELL", roundToTick(tpPartialPrice), na, na)

    else if currentSide == -1  // SHORT
        if high >= stopPrice and not tpPartialHit[1]
            hitStopLoss := true
        if tpPartialHit and high >= stopPrice and not hitStopLoss
            hitBreakeven := true

        if not na(tpFinalPrice) and low <= tpFinalPrice and not hitStopLoss and not hitBreakeven
            hitFinalExit := true

        if not tpPartialHit and not na(tpPartialPrice) and low <= tpPartialPrice and not hitStopLoss and not hitFinalExit
            hitTpPartial := true
            tpPartialHit := true
            stopPrice    := entryPrice
            if not na(lineStop)
                line.delete(lineStop)
            lineStop := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.orange, style=line.style_dotted, width=1)
            if not na(lineTpPartial)
                line.delete(lineTpPartial)
                lineTpPartial := na
            triggerAlert("PARTIAL", "BUY", roundToTick(tpPartialPrice), na, na)

    bool closePosition = hitStopLoss or hitBreakeven or hitFinalExit

    if closePosition
        string closeSide = currentSide == 1 ? "SELL" : "BUY"
        float  exitPrice = hitStopLoss ? stopPrice : (hitBreakeven ? entryPrice : tpFinalPrice)
        triggerAlert("CLOSE", closeSide, roundToTick(exitPrice), na, na)

        positionActive    := false
        currentSide       := 0
        firedCaseLabel    := na
        lastActionBar     := bar_index
        lastCloseBarIndex := bar_index
        tpPartialHit      := false
        if not na(lineEntry)
            line.delete(lineEntry)
            lineEntry := na
        if not na(lineStop)
            line.delete(lineStop)
            lineStop := na
        if not na(lineTpPartial)
            line.delete(lineTpPartial)
            lineTpPartial := na
        if not na(lineTpFinal)
            line.delete(lineTpFinal)
            lineTpFinal := na
    else
        if not na(lineEntry)
            line.set_xy1(lineEntry, bar_index, entryPrice)
            line.set_xy2(lineEntry, bar_index + 1, entryPrice)
        if not na(lineStop)
            line.set_xy1(lineStop, bar_index, stopPrice)
            line.set_xy2(lineStop, bar_index + 1, stopPrice)
        if not na(lineTpPartial)
            line.set_xy1(lineTpPartial, bar_index, tpPartialPrice)
            line.set_xy2(lineTpPartial, bar_index + 1, tpPartialPrice)
        if not na(lineTpFinal)
            line.set_xy1(lineTpFinal, bar_index, tpFinalPrice)
            line.set_xy2(lineTpFinal, bar_index + 1, tpFinalPrice)

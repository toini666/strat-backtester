roundToTick(float price) =>
    math.round(price / syminfo.mintick) * syminfo.mintick

getTimeframe() =>
    string tf = timeframe.period
    float tfNum = str.tonumber(tf)
    string result = if not na(tfNum)
        int tfMinutes = int(tfNum)
        tfMinutes >= 60 and tfMinutes % 60 == 0 ? "H" + str.tostring(tfMinutes / 60) : "M" + str.tostring(tfMinutes)
    else if str.endswith(tf, "S")
        "S" + str.substring(tf, 0, str.length(tf) - 1)
    else if tf == "D"
        "D1"
    else if str.endswith(tf, "D")
        "D" + str.substring(tf, 0, str.length(tf) - 1)
    else if tf == "W"
        "W1"
    else if str.endswith(tf, "W")
        "W" + str.substring(tf, 0, str.length(tf) - 1)
    else if tf == "M"
        "MO1"
    else if str.endswith(tf, "M")
        "MO" + str.substring(tf, 0, str.length(tf) - 1)
    else
        tf
    result

isInTimeSlot(int hh, int mm, int startH, int startM, int endH, int endM) =>
    int curMin   = hh * 60 + mm
    int startMin = startH * 60 + startM
    int endMin   = endH * 60 + endM
    bool endIsDayLastMinute = endH == 23 and endM == 59
    if startMin <= endMin
        curMin >= startMin and (endIsDayLastMinute ? curMin <= endMin : curMin < endMin)
    else
        curMin >= startMin or (endIsDayLastMinute ? curMin <= endMin : curMin < endMin)

isBlackoutWindow() =>
    int hh = hour(time_close, "Europe/Brussels")
    int mm = minute(time_close, "Europe/Brussels")
    bool b1 = i_bo1On and isInTimeSlot(hh, mm, i_bo1StartH, i_bo1StartM, i_bo1EndH, i_bo1EndM)
    bool b2 = i_bo2On and isInTimeSlot(hh, mm, i_bo2StartH, i_bo2StartM, i_bo2EndH, i_bo2EndM)
    bool b3 = i_bo3On and isInTimeSlot(hh, mm, i_bo3StartH, i_bo3StartM, i_bo3EndH, i_bo3EndM)
    bool b4 = i_bo4On and isInTimeSlot(hh, mm, i_bo4StartH, i_bo4StartM, i_bo4EndH, i_bo4EndM)
    bool b5 = i_bo5On and isInTimeSlot(hh, mm, i_bo5StartH, i_bo5StartM, i_bo5EndH, i_bo5EndM)
    bool b6 = i_bo6On and isInTimeSlot(hh, mm, i_bo6StartH, i_bo6StartM, i_bo6EndH, i_bo6EndM)
    b1 or b2 or b3 or b4 or b5 or b6

isAutoCloseBar() =>
    int openMin   = hour(time, "Europe/Brussels") * 60 + minute(time, "Europe/Brussels")
    int closeMin  = hour(time_close, "Europe/Brussels") * 60 + minute(time_close, "Europe/Brussels")
    int targetMin = i_autoCloseH * 60 + i_autoCloseM
    closeMin >= openMin ? (targetMin >= openMin and targetMin <= closeMin) : (targetMin >= openMin or targetMin <= closeMin)

method st(simple string src, float oscVal, simple int len) =>
    float o = switch src
        "SMA" => ta.sma(oscVal, len)
        "EMA" => ta.ema(oscVal, len)
    o

calcOscillator() =>
    float hi = ta.highest(mL)
    float lo = ta.lowest(mL)
    float av = ta.sma(hl2, mL)
    osc.sig := ta.ema(ta.linreg((close - math.avg(hi, lo, av)) / (hi - lo) * 100, mL, 0), sL)
    osc.sgD := sT.st(osc.sig, 2)

calcMFI() =>
    mf.mfi := currentMfi

updateMFITracking() =>
    if mf.mfi > 0
        if mf.brT.size() > 1
            mf.brT.pop()
        if mf.blT.size() > mfL
            mf.blT.pop()
        if mf.mfi > mf.blT.avg()
            mf.blT.unshift(mf.mfi)
        else
            float histMfi = currentMfi[mfL]
            mf.blT.unshift(not na(histMfi) and histMfi > 0 ? histMfi : mf.mfi)
    if mf.mfi < 0
        if mf.blT.size() > 1
            mf.blT.pop()
        if mf.brT.size() > mfL
            mf.brT.pop()
        if mf.mfi < mf.blT.avg()
            mf.brT.unshift(mf.mfi)
        else
            float histMfi = currentMfi[mfL]
            mf.brT.unshift(not na(histMfi) and histMfi < 0 ? histMfi : mf.mfi)

// Résultat oscillateur global
bool oscAllLongOk  = hwDirLongOk  and hwExtremeLongOk  and sigExtremeLongOk  and hwRangeOk and cloudOrDeltaLongOk  and deltaLongOk  and cloudZeroLongOk  and deltaExtLongOk
bool oscAllShortOk = hwDirShortOk and hwExtremeShortOk and sigExtremeShortOk and hwRangeOk and cloudOrDeltaShortOk and deltaShortOk and cloudZeroShortOk and deltaExtShortOk

// ============================================
// CONDITIONS D'ENTRÉE
// ============================================

// Setup V4 long:
//   - HMA lente passe de dessus à dessous la baseline SSL
//   - La fenêtre d'entrée reste ouverte pendant N bougies
// Setup V4 short:
//   - HMA lente passe de dessous à dessus la baseline SSL
//   - La fenêtre d'entrée reste ouverte pendant N bougies

bool slowHmaCrossLong  = hma2 < BBMC_ssl and hma2[1] > BBMC_ssl[1]
bool slowHmaCrossShort = hma2 > BBMC_ssl and hma2[1] < BBMC_ssl[1]
bool fastHmaExitLong   = hma1 > BBMC_ssl and hma1[1] < BBMC_ssl[1]
bool fastHmaExitShort  = hma1 < BBMC_ssl and hma1[1] > BBMC_ssl[1]

if slowHmaCrossLong
    lastLongSetupBar := bar_index
if slowHmaCrossShort
    lastShortSetupBar := bar_index

bool entryWindowLongOk  = not na(lastLongSetupBar)  and (bar_index - lastLongSetupBar)  <= i_entryWindowBars
bool entryWindowShortOk = not na(lastShortSetupBar) and (bar_index - lastShortSetupBar) <= i_entryWindowBars

bool hmaBullRecent = canalGreen and not na(lastHmaFlipUpBar) and (bar_index - lastHmaFlipUpBar) <= i_hmaPolBars
bool hmaBearRecent = not canalGreen and not na(lastHmaFlipDownBar) and (bar_index - lastHmaFlipDownBar) <= i_hmaPolBars
bool hmaLongOk     = not canalGreen or hmaBullRecent
bool hmaShortOk    = canalGreen or hmaBearRecent

// Filtre taille de bougie (open → close)
float candlePct    = close != 0 ? math.abs(close - open) / close * 100 : na
bool  candleSizeOk = i_maxCandlePct == 0.0 or (not na(candlePct) and candlePct <= i_maxCandlePct)

bool bothWindowsBlock = i_blockEntryIfBothWindows and entryWindowLongOk and entryWindowShortOk

bool longSignal  = entryWindowLongOk  and not bothWindowsBlock and hmaLongOk  and oscAllLongOk  and candleSizeOk
bool shortSignal = entryWindowShortOk and not bothWindowsBlock and hmaShortOk and oscAllShortOk and candleSizeOk

// ============================================
// CALCUL SL & TP
// ============================================

float slBuffer       = i_tickBuffer * syminfo.mintick
int bullHwAge = na(lastBullHwBar) ? na : bar_index - lastBullHwBar
int bearHwAge = na(lastBearHwBar) ? na : bar_index - lastBearHwBar
int bullHwSlBar = not na(bullHwAge) and bullHwAge > HW_SL_LOOKAROUND ? lastBullHwBar : prevBullHwBar
int bearHwSlBar = not na(bearHwAge) and bearHwAge > HW_SL_LOOKAROUND ? lastBearHwBar : prevBearHwBar
int bullHwSlAge = na(bullHwSlBar) ? na : bar_index - bullHwSlBar
int bearHwSlAge = na(bearHwSlBar) ? na : bar_index - bearHwSlBar
bool bullHwTradeSlReady = not na(bullHwSlAge) and bullHwSlAge > HW_SL_LOOKAROUND
bool bearHwTradeSlReady = not na(bearHwSlAge) and bearHwSlAge > HW_SL_LOOKAROUND
int bullHwSlLen = bullHwTradeSlReady ? bullHwSlAge + HW_SL_LOOKAROUND + 1 : 1
int bearHwSlLen = bearHwTradeSlReady ? bearHwSlAge + HW_SL_LOOKAROUND + 1 : 1
float bullHwRangeLow = ta.lowest(low, bullHwSlLen)
float bearHwRangeHigh = ta.highest(high, bearHwSlLen)
float bullHwSlBase = bullHwTradeSlReady ? bullHwRangeLow : na
float bearHwSlBase = bearHwTradeSlReady ? bearHwRangeHigh : na
float slLong         = bullHwSlBase - slBuffer
float slShort        = bearHwSlBase + slBuffer
float slDistLong  = close - slLong
float slDistShort = slShort - close

// On ignore le trade si le SL est incohérent (SL >= Prix pour Long ou SL <= Prix pour Short)
bool logicalSlLong  = not na(slLong) and slLong < close
bool logicalSlShort = not na(slShort) and slShort > close
bool entryNotAtSlExtremeLongOk  = not i_rejectEntryAtSlExtreme or (logicalSlLong  and not na(bullHwSlBase)  and low  > bullHwSlBase)
bool entryNotAtSlExtremeShortOk = not i_rejectEntryAtSlExtreme or (logicalSlShort and not na(bearHwSlBase) and high < bearHwSlBase)

bool  slLongOk    = logicalSlLong  and entryNotAtSlExtremeLongOk  and slDistLong  <= i_maxSlPoints
bool  slShortOk   = logicalSlShort and entryNotAtSlExtremeShortOk and slDistShort <= i_maxSlPoints

float calcEntryLong     = close
float calcStopLong      = roundToTick(slLong)

float calcEntryShort    = close
float calcStopShort     = roundToTick(slShort)

bool cooldownOk = (bar_index - lastCloseBarIndex) >= i_cooldownBars
bool oneTradeWindowLongOk  = not i_oneTradePerEntryWindow or na(lastLongTradedSetupBar)  or lastLongTradedSetupBar  != lastLongSetupBar
bool oneTradeWindowShortOk = not i_oneTradePerEntryWindow or na(lastShortTradedSetupBar) or lastShortTradedSetupBar != lastShortSetupBar

bool finalLongEntry  = longSignal  and slLongOk  and cooldownOk and oneTradeWindowLongOk
bool finalShortEntry = shortSignal and slShortOk and cooldownOk and oneTradeWindowShortOk

// hmaFlipUp et hmaFlipDown restent disponibles pour le traçage et les alertes indépendantes.

// ============================================
// GESTION DES POSITIONS
// ============================================

bool canAct = bar_index > lastActionBar

// --- 0. CLÔTURE FORCÉE AUTO-CLOSE ---
bool forceCloseTriggered = false
bool forceCloseInProfit  = false
int  forceCloseSide      = 0

if positionActive and i_autoCloseOn and autoCloseBar and canAct
    forceCloseTriggered := true
    forceCloseSide      := currentSide
    forceCloseInProfit  := currentSide == 1 ? close > entryPrice : close < entryPrice
    positionActive      := false
    currentSide         := 0
    lastActionBar       := bar_index
    lastCloseBarIndex   := bar_index
    tpPartialHit        := false
    pendingFinalExitLong  := false
    pendingFinalExitShort := false
    earlyExitFiredLong    := false
    earlyExitFiredShort   := false
    canAct              := false
    if not na(lineEntry)
        line.delete(lineEntry)
        lineEntry := na
    if not na(lineStop)
        line.delete(lineStop)
        lineStop := na

// --- 1. ENTRÉE LONGUE ---
if not positionActive and finalLongEntry and canAct and not inBlackout
    positionActive   := true
    currentSide      := 1
    entryPrice       := roundToTick(calcEntryLong)
    stopPrice        := roundToTick(calcStopLong)
    initialStopPrice := roundToTick(calcStopLong)
    lastLongTradedSetupBar := lastLongSetupBar
    lastActionBar    := bar_index
    tpPartialHit     := false
    // Signal de sortie déjà structurellement actif ? (HMA rapide déjà au-dessus de la baseline SSL pour un long)
    earlyExitFiredLong := hma1 > BBMC_ssl
    if earlyExitFiredLong and i_earlyExitFiredMode == "HW + RR"
        pendingFinalExitLong := true
    lineEntry := line.new(bar_index, entryPrice, bar_index + 1, entryPrice, color=color.blue,   style=line.style_dotted, width=1)
    lineStop  := line.new(bar_index, stopPrice,  bar_index + 1, stopPrice,  color=color.red,    style=line.style_dotted, width=1)
    alert('{"ticker":"' + syminfo.ticker + '","type":"SIGNAL","side":"BUY","entry":' + str.tostring(entryPrice) + ',"stop":' + str.tostring(stopPrice) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + getTimeframe() + '","timeframe":"' + getTimeframe() + '"}', alert.freq_once_per_bar_close)

// --- 2. ENTRÉE COURTE ---
else if not positionActive and finalShortEntry and canAct and not inBlackout
    positionActive   := true
    currentSide      := -1
    entryPrice       := roundToTick(calcEntryShort)
    stopPrice        := roundToTick(calcStopShort)
    initialStopPrice := roundToTick(calcStopShort)
    lastShortTradedSetupBar := lastShortSetupBar
    lastActionBar    := bar_index
    tpPartialHit     := false
    // Signal de sortie déjà structurellement actif ? (HMA rapide déjà sous la baseline SSL pour un short)
    earlyExitFiredShort := hma1 < BBMC_ssl
    if earlyExitFiredShort and i_earlyExitFiredMode == "HW + RR"
        pendingFinalExitShort := true
    lineEntry := line.new(bar_index, entryPrice, bar_index + 1, entryPrice, color=color.blue,   style=line.style_dotted, width=1)
    lineStop  := line.new(bar_index, stopPrice,  bar_index + 1, stopPrice,  color=color.red,    style=line.style_dotted, width=1)
    alert('{"ticker":"' + syminfo.ticker + '","type":"SIGNAL","side":"SELL","entry":' + str.tostring(entryPrice) + ',"stop":' + str.tostring(stopPrice) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + getTimeframe() + '","timeframe":"' + getTimeframe() + '"}', alert.freq_once_per_bar_close)

// --- 3. SORTIES ---
bool hitStopLoss  = false
bool hitCanalExit = false
bool hitBreakeven = false

bool exitedLong            = false
bool exitedShort           = false
bool wasStopLoss           = false
bool wasBreakeven          = false
bool wasCanalExit          = false
bool wasInProfit           = false

if positionActive and canAct and bar_index > lastActionBar
    if currentSide == 1  // LONG
        if low <= stopPrice and not tpPartialHit[1]
            hitStopLoss := true
        if tpPartialHit and low <= stopPrice and not hitStopLoss
            hitBreakeven := true

        // TP Final V4 : signal = cross HMA rapide / baseline SSL,
        // exécution = à la clôture du prochain HW où le RR atteint i_finalExitMinRr.
        // Si seuil non atteint, sortie REPORTÉE au HW suivant (et option BE).

        // Sorties alternatives quand le signal était déjà actif à l'entrée
        if earlyExitFiredLong and i_earlyExitFiredMode == "Clôture canal HMA inverse"
            if close < canalLower and not hitBreakeven and not hitStopLoss
                hitCanalExit := true
        else if earlyExitFiredLong and i_earlyExitFiredMode == "Prochain cross HMA lente"
            if slowHmaCrossShort and not hitBreakeven and not hitStopLoss
                hitCanalExit := true

        // Sortie standard HW + RR (incluant "HW + RR" mode early : pendingFinalExitLong déjà armé à l'entrée)
        if i_tpModeFastHmaHw and fastHmaExitLong
            pendingFinalExitLong := true
        bool exitCondLong = pendingFinalExitLong and (hwCrossOver or hwCrossUnder)
        float finalRiskLong   = entryPrice - initialStopPrice
        float finalRewardLong = close - entryPrice
        bool  finalRrLongOk   = i_finalExitMinRr == 0.0 or (finalRiskLong > 0 and finalRewardLong / finalRiskLong >= i_finalExitMinRr)
        bool  mfiReportLong   = i_reportTpIfMfiOk and close < entryPrice and cloudLongOk
        if exitCondLong and finalRrLongOk and not mfiReportLong and not hitBreakeven and not hitStopLoss and not hitCanalExit
            hitCanalExit := true
        else if exitCondLong and (not finalRrLongOk or mfiReportLong) and not hitBreakeven and not hitStopLoss and not hitCanalExit
            // Sortie reportée au prochain HW ; pendingFinalExitLong reste true.
            if i_moveToBeOnRejectedExit and not tpPartialHit and close > entryPrice and stopPrice < entryPrice
                stopPrice := entryPrice
                if not na(lineStop)
                    line.delete(lineStop)
                lineStop := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.orange, style=line.style_dotted, width=1)

        // TP : cross HMA lente (sens opposé) - sortie immédiate à la clôture
        if i_tpModeSlowHmaCross and slowHmaCrossShort and not hitBreakeven and not hitStopLoss and not hitCanalExit
            hitCanalExit := true

        // SL→BE sur cross HMA rapide (optionnel, si trade en positif, hors earlyExitFired)
        if i_moveToBeOnFastHmaCross and fastHmaExitLong and close > entryPrice and not earlyExitFiredLong and not tpPartialHit and stopPrice < entryPrice and not hitCanalExit and not hitBreakeven and not hitStopLoss
            tpPartialHit := true
            stopPrice    := entryPrice
            if not na(lineStop)
                line.delete(lineStop)
            lineStop := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.orange, style=line.style_dotted, width=1)

    else if currentSide == -1  // SHORT
        if high >= stopPrice and not tpPartialHit[1]
            hitStopLoss := true
        if tpPartialHit and high >= stopPrice and not hitStopLoss
            hitBreakeven := true

        // TP Final V4 : signal = cross HMA rapide / baseline SSL,
        // exécution = à la clôture du prochain HW où le RR atteint i_finalExitMinRr.
        // Si seuil non atteint, sortie REPORTÉE au HW suivant (et option BE).

        // Sorties alternatives quand le signal était déjà actif à l'entrée
        if earlyExitFiredShort and i_earlyExitFiredMode == "Clôture canal HMA inverse"
            if close > canalUpper and not hitBreakeven and not hitStopLoss
                hitCanalExit := true
        else if earlyExitFiredShort and i_earlyExitFiredMode == "Prochain cross HMA lente"
            if slowHmaCrossLong and not hitBreakeven and not hitStopLoss
                hitCanalExit := true

        // Sortie standard HW + RR (incluant "HW + RR" mode early : pendingFinalExitShort déjà armé à l'entrée)
        if i_tpModeFastHmaHw and fastHmaExitShort
            pendingFinalExitShort := true
        bool exitCondShort = pendingFinalExitShort and (hwCrossOver or hwCrossUnder)
        float finalRiskShort   = initialStopPrice - entryPrice
        float finalRewardShort = entryPrice - close
        bool  finalRrShortOk   = i_finalExitMinRr == 0.0 or (finalRiskShort > 0 and finalRewardShort / finalRiskShort >= i_finalExitMinRr)
        bool  mfiReportShort   = i_reportTpIfMfiOk and close > entryPrice and cloudShortOk
        if exitCondShort and finalRrShortOk and not mfiReportShort and not hitBreakeven and not hitStopLoss and not hitCanalExit
            hitCanalExit := true
        else if exitCondShort and (not finalRrShortOk or mfiReportShort) and not hitBreakeven and not hitStopLoss and not hitCanalExit
            // Sortie reportée au prochain HW ; pendingFinalExitShort reste true.
            if i_moveToBeOnRejectedExit and not tpPartialHit and close < entryPrice and stopPrice > entryPrice
                stopPrice := entryPrice
                if not na(lineStop)
                    line.delete(lineStop)
                lineStop := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.orange, style=line.style_dotted, width=1)

        // TP : cross HMA lente (sens opposé) - sortie immédiate à la clôture
        if i_tpModeSlowHmaCross and slowHmaCrossLong and not hitBreakeven and not hitStopLoss and not hitCanalExit
            hitCanalExit := true

        // SL→BE sur cross HMA rapide (optionnel, si trade en positif, hors earlyExitFired)
        if i_moveToBeOnFastHmaCross and fastHmaExitShort and close < entryPrice and not earlyExitFiredShort and not tpPartialHit and stopPrice > entryPrice and not hitCanalExit and not hitBreakeven and not hitStopLoss
            tpPartialHit := true
            stopPrice    := entryPrice
            if not na(lineStop)
                line.delete(lineStop)
            lineStop := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.orange, style=line.style_dotted, width=1)

    bool closePosition = hitStopLoss or hitBreakeven or hitCanalExit

    if closePosition
        exitedLong   := currentSide == 1
        exitedShort  := currentSide == -1
        wasStopLoss  := hitStopLoss
        wasBreakeven := hitBreakeven
        wasCanalExit := hitCanalExit
        wasInProfit  := currentSide == 1 ? close > entryPrice : close < entryPrice

        string closeSide      = currentSide == 1 ? "SELL" : "BUY"
        float  exitPrice      = hitStopLoss ? stopPrice : (hitBreakeven ? entryPrice : close)
        string closeAlertJson = '{"ticker":"' + syminfo.ticker + '","type":"CLOSE","side":"' + closeSide + '","entry":' + str.tostring(roundToTick(exitPrice)) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + getTimeframe() + '","timeframe":"' + getTimeframe() + '"}'
        if hitCanalExit
            alert(closeAlertJson, alert.freq_once_per_bar_close)
        else
            alert(closeAlertJson, alert.freq_once_per_bar)

        positionActive    := false
        currentSide       := 0
        lastActionBar     := bar_index
        lastCloseBarIndex := bar_index
        tpPartialHit      := false
        pendingFinalExitLong  := false
        pendingFinalExitShort := false
        earlyExitFiredLong    := false
        earlyExitFiredShort   := false
        if not na(lineEntry)
            line.delete(lineEntry)
            lineEntry := na
        if not na(lineStop)
            line.delete(lineStop)
            lineStop := na
    else
        if not na(lineEntry)
            line.set_x2(lineEntry, bar_index + 1)
        if not na(lineStop)
            line.set_xy1(lineStop, bar_index, stopPrice)
            line.set_xy2(lineStop, bar_index + 1, stopPrice)

    // ── En-têtes ──────────────────────────────────────────

    int barsSinceLongSetup  = na(lastLongSetupBar)  ? na : bar_index - lastLongSetupBar

    // ── Polarité HMA ───────────────────────────────────────
    string canalDir = canalGreen ? "HMA rouge ↑" : "HMA vert ↓"
    table.cell(tbl, 0, 3, "Polarité " + canalDir, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)

    // ── ① Sens HW ─────────────────────────────────────────
    string hwDirStr = na(lastConfirmedHWValue)
         ? "① Sens HW (—)"
         : lastConfirmedHW == 1
             ? "① Sens HW (▲ " + str.tostring(math.round(lastConfirmedHWValue, 1)) + ")"

    table.cell(tbl, 0, 4, hwDirStr, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)

    table.cell(tbl, 0, 5, hwExtStr, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)

    table.cell(tbl, 2, 5, i_hwExtremeOn ? f_icon(hwExtremeShortOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_hwExtremeOn ? f_bg(hwExtremeShortOk) : f_bgNa())

    // ── ③ Extrêmes SIG ───────────────────────────────────
    string sigExtStr = "③ SIG Extrême (±" + str.tostring(i_sigExtreme, "#") + ")"
         + (na(oscSig) ? " —" : " = " + str.tostring(math.round(oscSig, 1)))
    table.cell(tbl, 0, 6, sigExtStr, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 6, i_sigExtremeOn ? f_icon(sigExtremeLongOk)  : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_sigExtremeOn ? f_bg(sigExtremeLongOk)  : f_bgNa())
    table.cell(tbl, 2, 6, i_sigExtremeOn ? f_icon(sigExtremeShortOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_sigExtremeOn ? f_bg(sigExtremeShortOk) : f_bgNa())

    // ── ④ Range HW ────────────────────────────────────────
    string hwRngStr = "④ HW Range (±" + str.tostring(i_hwRange, "#") + ")"
         + (na(lastConfirmedHWValue) ? " —" : " |" + str.tostring(math.round(math.abs(lastConfirmedHWValue), 1)) + "|")
    table.cell(tbl, 0, 7, hwRngStr, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 7, i_hwRangeOn ? f_icon(hwRangeOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_hwRangeOn ? f_bg(hwRangeOk) : f_bgNa())
    table.cell(tbl, 2, 7, i_hwRangeOn ? f_icon(hwRangeOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_hwRangeOn ? f_bg(hwRangeOk) : f_bgNa())

    // ── ⑤ Nuage MFI ──────────────────────────────────────
    string mfiCurStr  = not na(activeMfiCloud)   ? str.tostring(math.round(activeMfiCloud,   1)) : "—"
    string mfiRefStr  = not na(activeMfiRefLine) ? str.tostring(math.round(activeMfiRefLine, 1)) : "—"
    string cloudStr   = "⑤ Nuage (" + mfiCurStr + " / " + mfiRefStr + ")"
    color cloudBgL = not i_cloudOn ? f_bgNa()
         : cloudLongOk                     ? color.new(color.green,  30)
         : (i_deltaOn and deltaLongOn)      ? color.new(color.orange, 30)
         : color.new(color.red, 30)
    color cloudBgS = not i_cloudOn ? f_bgNa()
         : cloudShortOk                    ? color.new(color.green,  30)
         : (i_deltaOn and deltaShortOn)     ? color.new(color.orange, 30)
         : color.new(color.red, 30)
    string cloudIconL = not i_cloudOn ? "—" : cloudLongOk  ? "✓" : (i_deltaOn and deltaLongOn)  ? "~" : "✗"
    string cloudIconS = not i_cloudOn ? "—" : cloudShortOk ? "✓" : (i_deltaOn and deltaShortOn) ? "~" : "✗"
    table.cell(tbl, 0, 8, cloudStr,   text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 8, cloudIconL, text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=cloudBgL)
    table.cell(tbl, 2, 8, cloudIconS, text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=cloudBgS)

    // ── ⑥ Deltas ─────────────────────────────────────────
    string sigValStr = not na(oscSig) ? str.tostring(math.round(oscSig, 1)) : "—"
    string mfiValStr = not na(activeMfiCloud) ? str.tostring(math.round(activeMfiCloud, 1)) : "—"
    string deltaStr  = "⑥ Deltas (sig " + sigValStr + " / mfi " + mfiValStr + ")"
    table.cell(tbl, 0, 9, deltaStr, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 9, i_deltaOn ? f_icon(deltaLongOk)  : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_deltaOn ? f_bg(deltaLongOk)  : f_bgNa())
    table.cell(tbl, 2, 9, i_deltaOn ? f_icon(deltaShortOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_deltaOn ? f_bg(deltaShortOk) : f_bgNa())

    // ── ⑦ Nuage < 0 / > 0 ────────────────────────────────
    string cloudZeroStr = "⑦ Nuage < 0 / > 0 (" + (not na(activeMfiCloud) ? str.tostring(math.round(activeMfiCloud, 1)) : "—") + ")"
    table.cell(tbl, 0, 10, cloudZeroStr, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 10, i_cloudZeroOn ? f_icon(cloudZeroLongOk)  : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_cloudZeroOn ? f_bg(cloudZeroLongOk)  : f_bgNa())
    table.cell(tbl, 2, 10, i_cloudZeroOn ? f_icon(cloudZeroShortOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_cloudZeroOn ? f_bg(cloudZeroShortOk) : f_bgNa())

    // ── ⑧ Extinction deltas ──────────────────────────────
    bool deltaShortWasOn = deltaShortOn[1]
    bool deltaLongWasOn  = deltaLongOn[1]
    string deltaExtLStr = "⑧ Ext. Δ (" + (deltaShortWasOn and not deltaShortOn ? "↓S" : deltaShortOn ? "S" : "—") + ")"
    string deltaExtSStr = "⑧ Ext. Δ (" + (deltaLongWasOn  and not deltaLongOn  ? "↓L" : deltaLongOn  ? "L" : "—") + ")"
    table.cell(tbl, 0, 11, "⑧ Extinction deltas", text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 11, i_deltaExtOn ? f_icon(deltaExtLongOk)  : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_deltaExtOn ? f_bg(deltaExtLongOk)  : f_bgNa())
    table.cell(tbl, 2, 11, i_deltaExtOn ? f_icon(deltaExtShortOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_deltaExtOn ? f_bg(deltaExtShortOk) : f_bgNa())

    // ── Distance SL ───────────────────────────────────────
    string slLongSrc  = "HW-" + str.tostring(bullHwSlAge) + "b"
    string slShortSrc = "HW-" + str.tostring(bearHwSlAge) + "b"
    string slLongStr  = na(bullHwSlBase) ? "Pas HW bull >2" : not logicalSlLong ? "Invalide (SL>Px)" : not entryNotAtSlExtremeLongOk ? "Bougie = extrême" : (na(slDistLong) ? "—" : slLongSrc + " " + str.tostring(slDistLong,  "#.#") + " pts")
    string slShortStr = na(bearHwSlBase) ? "Pas HW bear >2" : not logicalSlShort ? "Invalide (SL<Px)" : not entryNotAtSlExtremeShortOk ? "Bougie = extrême" : (na(slDistShort) ? "—" : slShortSrc + " " + str.tostring(slDistShort, "#.#") + " pts")
    table.cell(tbl, 0, 12, "SL depuis HW-2 (max " + str.tostring(i_maxSlPoints, "#") + ")", text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 12, slLongStr,  text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=f_bg(slLongOk))
    table.cell(tbl, 2, 12, slShortStr, text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=f_bg(slShortOk))

    // ── Filtre taille bougie ──────────────────────────────
    string candleStr = i_maxCandlePct == 0.0 ? "Bougie (OFF)" : "Bougie (max " + str.tostring(i_maxCandlePct, "#.##") + "%)"
    string candleVal = not na(candlePct) ? str.tostring(candlePct, "#.##") + "%" : "—"
    table.cell(tbl, 0, 13, candleStr + " = " + candleVal, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 13, i_maxCandlePct == 0.0 ? "—" : f_icon(candleSizeOk), text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_maxCandlePct == 0.0 ? f_bgNa() : f_bg(candleSizeOk))
    table.cell(tbl, 2, 13, i_maxCandlePct == 0.0 ? "—" : f_icon(candleSizeOk), text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_maxCandlePct == 0.0 ? f_bgNa() : f_bg(candleSizeOk))

    // ── Un trade max par fenêtre ──────────────────────────
    string oneTradeWindowStr = i_oneTradePerEntryWindow ? "Un trade/fenêtre" : "Un trade/fenêtre (OFF)"
    table.cell(tbl, 0, 14, oneTradeWindowStr, text_color=TXT_WHITE, text_size=i_tblSize, text_halign=text.align_left)
    table.cell(tbl, 1, 14, i_oneTradePerEntryWindow ? f_icon(oneTradeWindowLongOk)  : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_oneTradePerEntryWindow ? f_bg(oneTradeWindowLongOk)  : f_bgNa())
    table.cell(tbl, 2, 14, i_oneTradePerEntryWindow ? f_icon(oneTradeWindowShortOk) : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=i_oneTradePerEntryWindow ? f_bg(oneTradeWindowShortOk) : f_bgNa())

    // ── Résultat global ───────────────────────────────────
    table.cell(tbl, 0, 15, "RÉSULTAT",                                                              text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=HDR_BG, text_halign=text.align_left)
    table.cell(tbl, 1, 15, finalLongEntry  ? "✓ LONG"  : "✗", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=finalLongEntry  ? RES_BG_OK : RES_BG_KO)
    table.cell(tbl, 2, 15, finalShortEntry ? "✓ SHORT" : "✗", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=finalShortEntry ? RES_BG_OK : RES_BG_KO)

    // ── Position active ───────────────────────────────────
    string posStatus = positionActive ? (currentSide == 1 ? "LONG actif" : "SHORT actif") : "—"
    table.cell(tbl, 0, 16, "Position", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=HDR_BG, text_halign=text.align_left)
    table.cell(tbl, 1, 16, positionActive and currentSide == 1  ? posStatus : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=positionActive and currentSide == 1  ? color.new(color.green, 30) : f_bgNa())
    table.cell(tbl, 2, 16, positionActive and currentSide == -1 ? posStatus : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=positionActive and currentSide == -1 ? color.new(color.red,   30) : f_bgNa())

    // ── SL→BE sur cross HMA rapide ────────────────────────
    string beLabel  = i_moveToBeOnFastHmaCross ? "SL→BE cross HMA rapide" : "SL→BE cross HMA rapide (OFF)"
    string beStatus = tpPartialHit ? "✓ BE" : (positionActive and i_moveToBeOnFastHmaCross ? "attente cross HMA rapide" : "—")
    color  beColor  = tpPartialHit ? color.new(color.yellow, 35) : (positionActive and i_moveToBeOnFastHmaCross ? color.new(color.gray, 50) : f_bgNa())
    table.cell(tbl, 0, 17, beLabel, text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=HDR_BG, text_halign=text.align_left)
    table.cell(tbl, 1, 17, positionActive and currentSide == 1  ? beStatus : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=positionActive and currentSide == 1  ? beColor : f_bgNa())
    table.cell(tbl, 2, 17, positionActive and currentSide == -1 ? beStatus : "—", text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=positionActive and currentSide == -1 ? beColor : f_bgNa())

    // ── TP Final ──────────────────────────────────────────
    string tpModesStr = ""
    if i_tpModeFastHmaHw
        tpModesStr := "HMA→HW" + (i_finalExitMinRr > 0.0 ? " RR≥" + str.tostring(i_finalExitMinRr, "#.##") : "") + (i_reportTpIfMfiOk ? " +MFI" : "")
    if i_tpModeSlowHmaCross
        tpModesStr := tpModesStr == "" ? "HMA lente" : tpModesStr + " | HMA lente"
    if tpModesStr == ""
        tpModesStr := "—"
    string tpFinalLabel = "TP Final (" + tpModesStr + ")"

    string tpFinalStatusL = "—"
    string tpFinalStatusS = "—"
    color  tpFinalColorL  = f_bgNa()
    color  tpFinalColorS  = f_bgNa()

    if positionActive
        bool   tpFinalPending = (currentSide == 1 and pendingFinalExitLong) or (currentSide == -1 and pendingFinalExitShort)
        float  curRiskV4   = currentSide == 1 ? (entryPrice - initialStopPrice) : (initialStopPrice - entryPrice)
        float  curRewardV4 = currentSide == 1 ? (close - entryPrice) : (entryPrice - close)
        float  curRrV4     = (not na(curRiskV4) and curRiskV4 > 0) ? curRewardV4 / curRiskV4 : na
        string rrStrV4     = not na(curRrV4) ? str.tostring(curRrV4, "#.##") + "R" : "—"
        string s = not tpFinalPending ? "actif" :
                   i_finalExitMinRr > 0.0 ? "attente HW (" + rrStrV4 + ")" :
                   "attente HW"
        color  c = tpFinalPending ? color.new(color.orange, 35) : color.new(color.lime, 40)
        if currentSide == 1
            tpFinalStatusL := s
            tpFinalColorL  := c
        else if currentSide == -1
            tpFinalStatusS := s
            tpFinalColorS  := c

    table.cell(tbl, 0, 18, tpFinalLabel, text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=HDR_BG, text_halign=text.align_left)
    table.cell(tbl, 1, 18, tpFinalStatusL, text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=tpFinalColorL)
    table.cell(tbl, 2, 18, tpFinalStatusS, text_color=TXT_WHITE, text_size=i_tblSize, bgcolor=tpFinalColorS)

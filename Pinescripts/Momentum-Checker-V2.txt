roundToTick(float price) =>
    math.round(price / syminfo.mintick) * syminfo.mintick

getTimeframe() =>
    string tf = timeframe.period
    float tfNum = str.tonumber(tf)
    string result = if not na(tfNum)
        int tfMinutes = int(tfNum)
        tfMinutes >= 60 and tfMinutes % 60 == 0 ? "H" + str.tostring(tfMinutes / 60) : "M" + str.tostring(tfMinutes)
    else if str.endswith(tf, "S")
        "S" + str.substring(tf, 0, str.length(tf) - 1)
    else if tf == "D"
        "D1"
    else if str.endswith(tf, "D")
        "D" + str.substring(tf, 0, str.length(tf) - 1)
    else if tf == "W"
        "W1"
    else if str.endswith(tf, "W")
        "W" + str.substring(tf, 0, str.length(tf) - 1)
    else if tf == "M"
        "MO1"
    else if str.endswith(tf, "M")
        "MO" + str.substring(tf, 0, str.length(tf) - 1)
    else
        tf
    result

// --- HORAIRES (Timezone Bruxelles) ---
isInTimeSlot(int hh, int mm, int startH, int startM, int endH, int endM) =>
    int curMin   = hh * 60 + mm
    int startMin = startH * 60 + startM
    int endMin   = endH * 60 + endM
    bool endIsDayLastMinute = endH == 23 and endM == 59
    if startMin <= endMin
        curMin >= startMin and (endIsDayLastMinute ? curMin <= endMin : curMin < endMin)
    else
        curMin >= startMin or (endIsDayLastMinute ? curMin <= endMin : curMin < endMin)

isBlackoutWindow() =>
    int hh = hour(time_close, "Europe/Brussels")
    int mm = minute(time_close, "Europe/Brussels")
    bool b1 = i_bo1On and isInTimeSlot(hh, mm, i_bo1StartH, i_bo1StartM, i_bo1EndH, i_bo1EndM)
    bool b2 = i_bo2On and isInTimeSlot(hh, mm, i_bo2StartH, i_bo2StartM, i_bo2EndH, i_bo2EndM)
    bool b3 = i_bo3On and isInTimeSlot(hh, mm, i_bo3StartH, i_bo3StartM, i_bo3EndH, i_bo3EndM)
    bool b4 = i_bo4On and isInTimeSlot(hh, mm, i_bo4StartH, i_bo4StartM, i_bo4EndH, i_bo4EndM)
    bool b5 = i_bo5On and isInTimeSlot(hh, mm, i_bo5StartH, i_bo5StartM, i_bo5EndH, i_bo5EndM)
    bool b6 = i_bo6On and isInTimeSlot(hh, mm, i_bo6StartH, i_bo6StartM, i_bo6EndH, i_bo6EndM)
    b1 or b2 or b3 or b4 or b5 or b6

isAutoCloseBar() =>
    int curMin   = hour(time_close,    "Europe/Brussels") * 60 + minute(time_close,    "Europe/Brussels")
    int prevMin  = hour(time_close[1], "Europe/Brussels") * 60 + minute(time_close[1], "Europe/Brussels")
    int closeMin = i_autoCloseH * 60 + i_autoCloseM
    curMin >= closeMin and prevMin < closeMin

// Signal smoothing pour oscillateur
method st(simple string src, float oscVal, simple int len) =>
    float o = switch src
        "SMA" => ta.sma(oscVal, len)
        "EMA" => ta.ema(oscVal, len)
    o

// SMMA pour Alligator
smma(float src, int length) =>
    float s = 0.0
    float baseSma = ta.sma(src, length)
    s := na(s[1]) ? baseSma : (s[1] * (length - 1) + src) / length
    s

// 1) OSCILLATEUR (4Kings + Smart Money Flow)

type OscData
    float sig
    float sgD

type MFIData
    float   mfi
    float[] blT
    float[] brT

var OscData osc = OscData.new(na, na)
var MFIData mf  = MFIData.new(na, array.new<float>(1, na), array.new<float>(1, na))

// Variables non-var pour accès historique (TECHNICAL_MEMORY #1)
float oscSig = na
float oscSgD = na
float currentMfi = ta.sma(ta.mfi(hl2, mfL) - 50, mfS)

// Tracking dernière hyperwave confirmée
var int   lastConfirmedHW      = 0   // 1 = bullish, -1 = bearish
var float lastConfirmedHWValue = na

// Calcul oscillateur
calcOscillator() =>
    float hi = ta.highest(mL)
    float lo = ta.lowest(mL)
    float av = ta.sma(hl2, mL)
    osc.sig := ta.ema(ta.linreg((close - math.avg(hi, lo, av)) / (hi - lo) * 100, mL, 0), sL)
    osc.sgD := sT.st(osc.sig, 2)

calcMFI() =>
    mf.mfi := currentMfi

updateMFITracking() =>
    if mf.mfi > 0
        if mf.brT.size() > 1
            mf.brT.pop()
        if mf.blT.size() > mfL
            mf.blT.pop()
        if mf.mfi > mf.blT.avg()
            mf.blT.unshift(mf.mfi)
        else
            float histMfi = currentMfi[mfL]
            mf.blT.unshift(not na(histMfi) and histMfi > 0 ? histMfi : mf.mfi)

    if mf.mfi < 0
        if mf.blT.size() > 1
            mf.blT.pop()
        if mf.brT.size() > mfL
            mf.brT.pop()
        if mf.mfi < mf.blT.avg()
            mf.brT.unshift(mf.mfi)
        else
            float histMfi = currentMfi[mfL]
            mf.brT.unshift(not na(histMfi) and histMfi < 0 ? histMfi : mf.mfi)

calcOscillator()
calcMFI()

oscSig := osc.sig
oscSgD := osc.sgD

updateMFITracking()

// Détection Hyperwave confirmée (TECHNICAL_MEMORY #9: au scope global)
bool hwCrossOver  = ta.crossover(oscSig, oscSgD)
bool hwCrossUnder = ta.crossunder(oscSig, oscSgD)

if hwCrossOver
    lastConfirmedHW      := 1
    lastConfirmedHWValue := math.min(oscSig, oscSgD)
if hwCrossUnder
    lastConfirmedHW      := -1
    lastConfirmedHWValue := math.max(oscSig, oscSgD)

// Conditions oscillateur (tous les filtres sont scoring-style: OFF = 0 point + retiré de maxPts)

// HW Sens (toujours actif quand oscillateur ON, pas de toggle dédié)
bool oscHWSensLong   = i_oscOn and lastConfirmedHW == 1
bool oscHWSensShort  = i_oscOn and lastConfirmedHW == -1

// HW Range (valeur HW hors de la zone -N/+N)
bool oscHWValueOk = i_oscOn and i_hwFilterOn and not na(lastConfirmedHWValue) and math.abs(lastConfirmedHWValue) > i_hwLevel

// HW Extrême (valeur au dernier cross dans la zone autorisée)
bool oscHWExtremeLongOk  = i_oscOn and i_hwExtremeFilterOn and not na(lastConfirmedHWValue) and lastConfirmedHWValue <= i_hwExtreme
bool oscHWExtremeShortOk = i_oscOn and i_hwExtremeFilterOn and not na(lastConfirmedHWValue) and lastConfirmedHWValue >= -i_hwExtreme

// SIG Range (valeur SIG courante hors de la zone -N/+N)
bool oscSigValueOk = i_oscOn and i_sigFilterOn and not na(oscSig) and math.abs(oscSig) > i_sigLevel

// SIG Extrême courant (signal courant dans la zone autorisée)
bool oscSigExtremeLongOk  = i_oscOn and i_sigExtremeFilterOn and not na(oscSig) and oscSig <= i_sigExtreme
bool oscSigExtremeShortOk = i_oscOn and i_sigExtremeFilterOn and not na(oscSig) and oscSig >= -i_sigExtreme

// Deltas (osc + MFI alignés)
bool deltaLongOn   = osc.sig > 0 and mf.mfi > 0
bool deltaShortOn  = osc.sig < 0 and mf.mfi < 0
bool deltaOff      = not deltaLongOn and not deltaShortOn
// Bonus delta selon le mode (cf. i_deltaOffMode):
//   "both"         → si les deux deltas sont OFF, les deux côtés reçoivent le bonus (comportement V1)
//   "counter_trend"→ si OFF, seul le côté contre-tendance reçoit le bonus (MFI<0 → long, MFI>0 → short)
bool deltaLongBonus  = i_deltaOffMode == "both" ? (deltaLongOn  or deltaOff) : (deltaLongOn  or (deltaOff and not na(mf.mfi) and mf.mfi < 0))
bool deltaShortBonus = i_deltaOffMode == "both" ? (deltaShortOn or deltaOff) : (deltaShortOn or (deltaOff and not na(mf.mfi) and mf.mfi > 0))
bool oscDeltaLong  = i_oscOn and i_deltaFilterOn and deltaLongBonus
bool oscDeltaShort = i_oscOn and i_deltaFilterOn and deltaShortBonus

// Nuage MFI (position MFI vs sa référence dynamique)
float activeMfiRefLine   = mf.mfi > 0 ? mf.blT.avg() : mf.mfi < 0 ? mf.brT.avg() : na
float activeMfiCloudLine = mf.mfi
bool hasActiveMfiCloud   = not na(activeMfiCloudLine) and not na(activeMfiRefLine) and activeMfiCloudLine != 0
bool cloudLongAllowed    = hasActiveMfiCloud and activeMfiRefLine < activeMfiCloudLine
bool cloudShortAllowed   = hasActiveMfiCloud and activeMfiRefLine > activeMfiCloudLine
bool oscCloudLong  = i_oscOn and i_cloudFilterOn and cloudLongAllowed
bool oscCloudShort = i_oscOn and i_cloudFilterOn and cloudShortAllowed

// Nuage < 0 / > 0 (aligné v3)
bool oscCloudZeroLong  = i_oscOn and i_cloudZeroFilterOn and not na(mf.mfi) and mf.mfi < 0
bool oscCloudZeroShort = i_oscOn and i_cloudZeroFilterOn and not na(mf.mfi) and mf.mfi > 0

// ============================================
// 2) DOUBLE EMA
// ============================================
float emaPrin = ta.ema(close, i_emaPrinLen)
float emaSec  = ta.ema(close, i_emaSecLen)

// Break EMA principale: bougie courante ou 2 précédentes
// Long: open < ema, close > ema (bougie haussière qui casse l'EMA)
// Short: open > ema, close < ema (bougie baissière qui casse l'EMA)
bool emaBreakLong0  = close > open and open < emaPrin and close > emaPrin
bool emaBreakShort0 = close < open and open > emaPrin and close < emaPrin
bool emaBreakLong1  = close[1] > open[1] and open[1] < emaPrin[1] and close[1] > emaPrin[1]
bool emaBreakShort1 = close[1] < open[1] and open[1] > emaPrin[1] and close[1] < emaPrin[1]
bool emaBreakLong2  = close[2] > open[2] and open[2] < emaPrin[2] and close[2] > emaPrin[2]
bool emaBreakShort2 = close[2] < open[2] and open[2] > emaPrin[2] and close[2] < emaPrin[2]

bool emaBreakLong  = i_emaOn and (emaBreakLong0 or emaBreakLong1 or emaBreakLong2)
bool emaBreakShort = i_emaOn and (emaBreakShort0 or emaBreakShort1 or emaBreakShort2)

// EMA Alignment
bool emaAlignLong  = i_emaOn and emaSec > emaPrin
bool emaAlignShort = i_emaOn and emaSec < emaPrin

// ============================================
// 3) SUPERTREND
// ============================================
float stATR = ta.atr(i_stATR)
float stSrc = hl2

float stUp  = stSrc - (i_stMult * stATR)
float stUp1 = nz(stUp[1], stUp)
stUp := close[1] > stUp1 ? math.max(stUp, stUp1) : stUp

float stDn  = stSrc + (i_stMult * stATR)
float stDn1 = nz(stDn[1], stDn)
stDn := close[1] < stDn1 ? math.min(stDn, stDn1) : stDn

var int stTrend = 1
stTrend := stTrend == -1 and close > stDn1 ? 1 : stTrend == 1 and close < stUp1 ? -1 : stTrend

bool stBullish = i_stOn and stTrend == 1
bool stBearish = i_stOn and stTrend == -1

// ============================================
// 4) ALLIGATOR
// ============================================
float jaw   = smma(hl2, i_jawLen)
float teeth = smma(hl2, i_teethLen)
float lips  = smma(hl2, i_lipsLen)

// Valeurs affichées sur la bougie courante (avec offset)
float jawOnChart   = i_jawOff > 0 ? jaw[i_jawOff] : jaw
float teethOnChart = i_teethOff > 0 ? teeth[i_teethOff] : teeth
float lipsOnChart  = i_lipsOff > 0 ? lips[i_lipsOff] : lips

// Alligator bougie courante
bool alliChartBull = i_alOn and not na(jawOnChart) and not na(teethOnChart) and not na(lipsOnChart) and lipsOnChart > teethOnChart and teethOnChart > jawOnChart
bool alliChartBear = i_alOn and not na(jawOnChart) and not na(teethOnChart) and not na(lipsOnChart) and lipsOnChart < teethOnChart and teethOnChart < jawOnChart

// Alligator offset
bool alliOffsetBull = i_alOn and lips > teeth and teeth > jaw
bool alliOffsetBear = i_alOn and lips < teeth and teeth < jaw

// Retest Lips: alligator aligné (valeurs courantes) + bougie touche les lèvres + clôture de l'autre côté des lèvres
bool retestLipsLong  = i_alOn and alliChartBull and low <= lipsOnChart and close > lipsOnChart
bool retestLipsShort = i_alOn and alliChartBear and high >= lipsOnChart and close < lipsOnChart

// ============================================
// 5) UT BOT
// ============================================
float utXATR  = ta.atr(i_utATR)
float utNLoss = i_utKey * utXATR

float utSrc = close

var float xATRTrailingStop = na
xATRTrailingStop := utSrc > nz(xATRTrailingStop[1], 0) and utSrc[1] > nz(xATRTrailingStop[1], 0) ?
   math.max(nz(xATRTrailingStop[1]), utSrc - utNLoss) :
   utSrc < nz(xATRTrailingStop[1], 0) and utSrc[1] < nz(xATRTrailingStop[1], 0) ?
   math.min(nz(xATRTrailingStop[1]), utSrc + utNLoss) :
   utSrc > nz(xATRTrailingStop[1], 0) ? utSrc - utNLoss : utSrc + utNLoss

var int utPos = 0
utPos := utSrc[1] < nz(xATRTrailingStop[1], 0) and utSrc > nz(xATRTrailingStop[1], 0) ? 1 :
   utSrc[1] > nz(xATRTrailingStop[1], 0) and utSrc < nz(xATRTrailingStop[1], 0) ? -1 :
   nz(utPos[1], 0)

float utEma   = ta.ema(utSrc, 1)
bool  utAbove = ta.crossover(utEma, xATRTrailingStop)
bool  utBelow = ta.crossover(xATRTrailingStop, utEma)

bool utBotBuy  = i_utOn and utSrc > xATRTrailingStop and utAbove
bool utBotSell = i_utOn and utSrc < xATRTrailingStop and utBelow

// ============================================
// 6) STC
// ============================================
calcMACD(float source, int fastLen, int slowLen) =>
    fastMA = ta.ema(source, fastLen)
    slowMA = ta.ema(source, slowLen)
    fastMA - slowMA

calcSTC(int length, int fastLen, int slowLen) =>
    factor = 0.5
    var float pff = 0.0
    var float pf = 0.0
    var float pfff = 0.0
    var float pff2 = 0.0
    macdVal = calcMACD(close, fastLen, slowLen)
    lowest = ta.lowest(macdVal, length)
    highest = ta.highest(macdVal, length)
    range1 = highest - lowest
    pff := range1 > 0 ? (macdVal - lowest) / range1 * 100 : nz(pff[1])
    pf := na(pf[1]) ? pff : pf[1] + factor * (pff - pf[1])
    lowest2 = ta.lowest(pf, length)
    highest2 = ta.highest(pf, length)
    range2 = highest2 - lowest2
    pfff := range2 > 0 ? (pf - lowest2) / range2 * 100 : nz(pfff[1])
    pff2 := na(pff2[1]) ? pfff : pff2[1] + factor * (pfff - pff2[1])
    pff2

float stcValue = calcSTC(i_stcLength, i_stcFastLen, i_stcSlowLen)
bool stcRising  = stcValue > stcValue[1]
bool stcFalling = stcValue < stcValue[1]

bool stcLong  = i_stcOn and stcValue >= i_stcMinLong and stcValue <= i_stcMaxLong and stcRising
bool stcShort = i_stcOn and stcValue >= i_stcMinShort and stcValue <= i_stcMaxShort and stcFalling

// ============================================
// 7) HMA RIBBON + SSL
// ============================================

// --- SSL (BBMC + Keltner channel, comme v3) ---
float BBMC_ssl    = ta.wma(2 * ta.wma(close, i_sslLen / 2) - ta.wma(close, i_sslLen), math.round(math.sqrt(i_sslLen)))
float rangema_ssl = ta.ema(ta.tr, i_sslLen)
float upperk_ssl  = BBMC_ssl + rangema_ssl * i_sslMult
float lowerk_ssl  = BBMC_ssl - rangema_ssl * i_sslMult
color color_ssl   = close > upperk_ssl ? color.new(#00c3ff, 0) : close < lowerk_ssl ? color.new(#ff0062, 0) : color.gray

// --- HMA Ribbon ---
float hmaEma    = ta.ema(close, i_hmaEmaLen)
float rawHma1   = ta.hma(hmaEma, i_hma1Len)
float rawHma2   = ta.hma(hmaEma, i_hma2Len)
float hma1      = hmaEma + (rawHma1 - hmaEma) * i_ampMult
float hma2      = hmaEma + (rawHma2 - hmaEma) * i_ampMult

bool  canalGreen = hma1 > hma2
float canalUpper = math.max(hma1, hma2)
float canalLower = math.min(hma1, hma2)

// Flip canal HMA
bool hmaFlipUp   = ta.crossover(hma1, hma2)
bool hmaFlipDown = ta.crossunder(hma1, hma2)

var int lastHmaFlipUpBar   = na
var int lastHmaFlipDownBar = na

if hmaFlipUp
    lastHmaFlipUpBar := bar_index
if hmaFlipDown
    lastHmaFlipDownBar := bar_index

// Polarité récente: tolère le Break même si canal pas dans le bon sens, si flip récent
bool hmaBullRecent = not na(lastHmaFlipUpBar)   and (bar_index - lastHmaFlipUpBar)   <= i_hmaPolBars
bool hmaBearRecent = not na(lastHmaFlipDownBar) and (bar_index - lastHmaFlipDownBar) <= i_hmaPolBars
bool hmaPolLongOk  = canalGreen or hmaBullRecent
bool hmaPolShortOk = not canalGreen or hmaBearRecent

// Break HMA canal: close au-dessus/dessous du canal + polarité OK
bool hmaBreakLong  = i_hmaOn and hmaPolLongOk  and close > canalUpper
bool hmaBreakShort = i_hmaOn and hmaPolShortOk and close < canalLower

// Cross HMA lente/SSL (hma2 vs SSL baseline) + fenêtre d'entrée
// Direction v3 (counter-trend): hma2 croise EN-DESSOUS SSL = setup LONG, croise AU-DESSUS = setup SHORT
bool slowHmaCrossUp   = ta.crossover(hma2, BBMC_ssl)
bool slowHmaCrossDown = ta.crossunder(hma2, BBMC_ssl)
var int lastSlowHmaUpBar   = na
var int lastSlowHmaDownBar = na
if slowHmaCrossUp
    lastSlowHmaUpBar := bar_index
if slowHmaCrossDown
    lastSlowHmaDownBar := bar_index

bool hmaSlowCrossLong  = i_hmaOn and not na(lastSlowHmaDownBar) and (bar_index - lastSlowHmaDownBar) <= i_hmaWindowBars
bool hmaSlowCrossShort = i_hmaOn and not na(lastSlowHmaUpBar)   and (bar_index - lastSlowHmaUpBar)   <= i_hmaWindowBars

// ============================================

int ptsLong = 0
ptsLong += oscHWSensLong       ? i_ptsHWSens     : 0
ptsLong += oscHWValueOk        ? i_ptsHWValue    : 0
ptsLong += oscHWExtremeLongOk  ? i_ptsHWExtreme  : 0
ptsLong += oscSigValueOk       ? i_ptsSigValue   : 0
ptsLong += oscSigExtremeLongOk ? i_ptsSigExtreme : 0
ptsLong += oscCloudLong        ? i_ptsCloud     : 0
ptsLong += oscDeltaLong        ? i_ptsDelta     : 0
ptsLong += oscCloudZeroLong    ? i_ptsCloudZero : 0
ptsLong += emaBreakLong        ? i_ptsEmaBreak  : 0
ptsLong += emaAlignLong        ? i_ptsEmaAlign  : 0
ptsLong += stBullish           ? i_ptsST        : 0
ptsLong += alliChartBull       ? i_ptsAlligator : 0
ptsLong += alliOffsetBull      ? i_ptsAlliOffset: 0
ptsLong += retestLipsLong      ? i_ptsRetestLips: 0
ptsLong += utBotBuy            ? i_ptsUTBot     : 0
ptsLong += stcLong             ? i_ptsSTC       : 0
ptsLong += hmaBreakLong        ? i_ptsHMABreak  : 0
ptsLong += hmaSlowCrossLong    ? i_ptsHMASlow   : 0

// Points Short
int ptsShort = 0
ptsShort += oscHWSensShort      ? i_ptsHWSens     : 0
ptsShort += oscHWValueOk        ? i_ptsHWValue    : 0
ptsShort += oscHWExtremeShortOk ? i_ptsHWExtreme  : 0
ptsShort += oscSigValueOk       ? i_ptsSigValue   : 0
ptsShort += oscSigExtremeShortOk? i_ptsSigExtreme : 0
ptsShort += oscCloudShort       ? i_ptsCloud     : 0
ptsShort += oscDeltaShort       ? i_ptsDelta     : 0
ptsShort += oscCloudZeroShort   ? i_ptsCloudZero : 0
ptsShort += emaBreakShort       ? i_ptsEmaBreak  : 0
ptsShort += emaAlignShort       ? i_ptsEmaAlign  : 0
ptsShort += stBearish           ? i_ptsST        : 0
ptsShort += alliChartBear       ? i_ptsAlligator : 0
ptsShort += alliOffsetBear      ? i_ptsAlliOffset: 0
ptsShort += retestLipsShort     ? i_ptsRetestLips: 0
ptsShort += utBotSell           ? i_ptsUTBot     : 0
ptsShort += stcShort            ? i_ptsSTC       : 0
ptsShort += hmaBreakShort       ? i_ptsHMABreak  : 0
ptsShort += hmaSlowCrossShort   ? i_ptsHMASlow   : 0

// Points maximum théorique
int maxOscPts = i_oscOn ? i_ptsHWSens + (i_hwFilterOn ? i_ptsHWValue : 0) + (i_hwExtremeFilterOn ? i_ptsHWExtreme : 0) + (i_sigFilterOn ? i_ptsSigValue : 0) + (i_sigExtremeFilterOn ? i_ptsSigExtreme : 0) + (i_cloudFilterOn ? i_ptsCloud : 0) + (i_deltaFilterOn ? i_ptsDelta : 0) + (i_cloudZeroFilterOn ? i_ptsCloudZero : 0) : 0
int maxPtsLong  = maxOscPts + (i_emaOn ? i_ptsEmaBreak + i_ptsEmaAlign : 0) + (i_stOn ? i_ptsST : 0) + (i_alOn ? i_ptsAlligator + i_ptsAlliOffset + i_ptsRetestLips : 0) + (i_utOn ? i_ptsUTBot : 0) + (i_stcOn ? i_ptsSTC : 0) + (i_hmaOn ? i_ptsHMABreak + i_ptsHMASlow : 0)
int maxPtsShort = maxPtsLong

// CONDITIONS D'ENTRÉE
// ============================================
bool gapLongOk  = (ptsLong  - ptsShort) >= i_minGap
bool gapShortOk = (ptsShort - ptsLong)  >= i_minGap
bool longEntry  = ptsLong  >= i_longThreshold  and gapLongOk  and candleSizeOk and not inBlackout
bool shortEntry = ptsShort >= i_shortThreshold and gapShortOk and candleSizeOk and not inBlackout

// Calcul des niveaux
// Le risk subit (dans l'ordre) : élargissement au min(% du prix) puis plafond à i_slMaxPoints.
// Si min > max, le plafond gagne (config-error).
float calcEntryLong  = close
float calcStopLong   = ta.lowest(low, i_slLookback) - (i_tickBuffer * syminfo.mintick)
float minPctRiskL    = calcEntryLong  * (i_slMinPct / 100.0)
float riskL          = math.min(math.max(calcEntryLong - calcStopLong, minPctRiskL), i_slMaxPoints)
calcStopLong        := calcEntryLong - riskL
float calcTpLong     = calcEntryLong + (riskL * i_rrTP)

float calcEntryShort = close
float calcStopShort  = ta.highest(high, i_slLookback) + (i_tickBuffer * syminfo.mintick)
float minPctRiskS    = calcEntryShort * (i_slMinPct / 100.0)
float riskS          = math.min(math.max(calcStopShort - calcEntryShort, minPctRiskS), i_slMaxPoints)
calcStopShort       := calcEntryShort + riskS
float calcTpShort    = calcEntryShort - (riskS * i_rrTP)

// ============================================
// GESTION ET EXÉCUTION DES POSITIONS
// ============================================
bool canAct = bar_index > lastActionBar
bool forceCloseTriggered = false
int  forceCloseSide      = 0

// 0. AUTO-CLOSE
if positionActive and autoCloseBar and canAct
    forceCloseTriggered := true
    forceCloseSide      := currentSide
    positionActive := false
    currentSide    := 0
    beTriggered    := false
    lastActionBar  := bar_index
    canAct         := false
    if not na(lineEntry)
        line.delete(lineEntry)
        lineEntry := na
    if not na(lineStop)
        line.delete(lineStop)
        lineStop := na
    if not na(lineTP)
        line.delete(lineTP)
        lineTP := na

// 1. ENTRÉE LONGUE
if not positionActive and longEntry and canAct
    positionActive   := true
    currentSide      := 1
    entryPrice       := roundToTick(calcEntryLong)
    stopPrice        := roundToTick(calcStopLong)
    tpPrice          := roundToTick(calcTpLong)
    initialStopPrice := stopPrice
    beTriggered      := false
    lastActionBar    := bar_index

    lineEntry := line.new(bar_index, entryPrice, bar_index + 1, entryPrice, color=color.blue, style=line.style_dotted, width=1)
    lineStop  := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.red, style=line.style_dotted, width=1)
    lineTP    := line.new(bar_index, tpPrice, bar_index + 1, tpPrice, color=color.green, style=line.style_dotted, width=1)

    string alertTf = getTimeframe()
    alert('{"ticker":"' + syminfo.ticker + '","type":"SIGNAL","side":"BUY","entry":' + str.tostring(entryPrice) + ',"stop":' + str.tostring(stopPrice) + ',"tp":' + str.tostring(tpPrice) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + alertTf + '","timeframe":"' + alertTf + '"}', alert.freq_once_per_bar_close)

// 2. ENTRÉE COURTE
else if not positionActive and shortEntry and canAct
    positionActive   := true
    currentSide      := -1
    entryPrice       := roundToTick(calcEntryShort)
    stopPrice        := roundToTick(calcStopShort)
    tpPrice          := roundToTick(calcTpShort)
    initialStopPrice := stopPrice
    beTriggered      := false
    lastActionBar    := bar_index

    lineEntry := line.new(bar_index, entryPrice, bar_index + 1, entryPrice, color=color.blue, style=line.style_dotted, width=1)
    lineStop  := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.red, style=line.style_dotted, width=1)
    lineTP    := line.new(bar_index, tpPrice, bar_index + 1, tpPrice, color=color.green, style=line.style_dotted, width=1)

    string alertTf = getTimeframe()
    alert('{"ticker":"' + syminfo.ticker + '","type":"SIGNAL","side":"SELL","entry":' + str.tostring(entryPrice) + ',"stop":' + str.tostring(stopPrice) + ',"tp":' + str.tostring(tpPrice) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + alertTf + '","timeframe":"' + alertTf + '"}', alert.freq_once_per_bar_close)

// 2.5. PASSAGE AU BREAK-EVEN (pas sur la bougie d'entrée: sinon low <= entry déclencherait une sortie BE immédiate)
if positionActive and bar_index > lastActionBar and i_beAtRR > 0.0 and not beTriggered
    float initialRiskL = entryPrice - initialStopPrice
    float initialRiskS = initialStopPrice - entryPrice
    bool beHitLong  = currentSide == 1  and initialRiskL > 0 and (high - entryPrice) >= initialRiskL * i_beAtRR
    bool beHitShort = currentSide == -1 and initialRiskS > 0 and (entryPrice - low)  >= initialRiskS * i_beAtRR
    if beHitLong or beHitShort
        beTriggered := true
        stopPrice   := entryPrice
        if not na(lineStop)
            line.delete(lineStop)
        lineStop := line.new(bar_index, stopPrice, bar_index + 1, stopPrice, color=color.orange, style=line.style_dotted, width=1)

        string alertTfBE = getTimeframe()
        string beSide    = currentSide == 1 ? "BUY" : "SELL"
        alert('{"ticker":"' + syminfo.ticker + '","type":"MOVEBE","side":"' + beSide + '","entry":' + str.tostring(entryPrice) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + alertTfBE + '","timeframe":"' + alertTfBE + '"}', alert.freq_once_per_bar)

// 3. SORTIES
bool exitedLong  = false
bool exitedShort = false
bool wasTP       = false
bool wasBE       = false

if positionActive and bar_index > lastActionBar
    bool closeLong  = currentSide == 1 and (low <= stopPrice or high >= tpPrice)
    bool closeShort = currentSide == -1 and (high >= stopPrice or low <= tpPrice)

    if closeLong
        wasTP      := high >= tpPrice
        wasBE      := not wasTP and beTriggered and stopPrice == entryPrice
        exitedLong := true
        float exitPrice = wasTP ? tpPrice : stopPrice
        string alertTf = getTimeframe()
        alert('{"ticker":"' + syminfo.ticker + '","type":"CLOSE","side":"SELL","entry":' + str.tostring(roundToTick(exitPrice)) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + alertTf + '","timeframe":"' + alertTf + '"}', alert.freq_once_per_bar)
        positionActive := false
        currentSide    := 0
        beTriggered    := false
        lastActionBar  := bar_index
        if not na(lineEntry)
            line.delete(lineEntry)
            lineEntry := na
        if not na(lineStop)
            line.delete(lineStop)
            lineStop := na
        if not na(lineTP)
            line.delete(lineTP)
            lineTP := na

    else if closeShort
        wasTP       := low <= tpPrice
        wasBE       := not wasTP and beTriggered and stopPrice == entryPrice
        exitedShort := true
        float exitPrice = wasTP ? tpPrice : stopPrice
        string alertTf = getTimeframe()
        alert('{"ticker":"' + syminfo.ticker + '","type":"CLOSE","side":"BUY","entry":' + str.tostring(roundToTick(exitPrice)) + ',"strat":"' + i_stratName + '-' + syminfo.ticker + '-' + alertTf + '","timeframe":"' + alertTf + '"}', alert.freq_once_per_bar)
        positionActive := false
        currentSide    := 0
        beTriggered    := false
        lastActionBar  := bar_index
        if not na(lineEntry)
            line.delete(lineEntry)
            lineEntry := na
        if not na(lineStop)
            line.delete(lineStop)
            lineStop := na
        if not na(lineTP)
            line.delete(lineTP)
            lineTP := na

    else
        // Extension des lignes
        if not na(lineEntry)
            line.set_xy1(lineEntry, bar_index, entryPrice)
            line.set_xy2(lineEntry, bar_index + 1, entryPrice)
        if not na(lineStop)
            line.set_xy1(lineStop, bar_index, stopPrice)
            line.set_xy2(lineStop, bar_index + 1, stopPrice)
        if not na(lineTP)
            line.set_xy1(lineTP, bar_index, tpPrice)
            line.set_xy2(lineTP, bar_index + 1, tpPrice)

    table.cell(statusTable, 0, row, "TOTAL", text_color=color.white, bgcolor=color.new(color.gray, 60), text_size=size.small)
    table.cell(statusTable, 1, row, "Prép/Décl: " + str.tostring(i_longPrepThreshold) + "-" + str.tostring(i_longThreshold) + "/" + str.tostring(i_shortPrepThreshold) + "-" + str.tostring(i_shortThreshold), text_color=color.white, bgcolor=color.new(color.gray, 60), text_size=size.small)
    table.cell(statusTable, 2, row, str.tostring(ptsLong) + "/" + str.tostring(maxPtsLong), text_color=color.white, bgcolor=color.new(totalLongColor, 60), text_size=size.small)
    table.cell(statusTable, 3, row, str.tostring(ptsShort) + "/" + str.tostring(maxPtsShort), text_color=color.white, bgcolor=color.new(totalShortColor, 60), text_size=size.small)
    row += 1

    // --- Écart L/S ---
    int gapL = ptsLong  - ptsShort
    int gapS = ptsShort - ptsLong
    color gapLColor = gapLongOk  ? color.green : color.gray
    color gapSColor = gapShortOk ? color.red   : color.gray
    table.cell(statusTable, 0, row, "Écart L/S", text_color=color.white, bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 1, row, "Min: " + str.tostring(i_minGap), text_color=color.white, bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 2, row, str.tostring(gapL), text_color=color.white, bgcolor=color.new(gapLColor, 70), text_size=size.small)
    table.cell(statusTable, 3, row, str.tostring(gapS), text_color=color.white, bgcolor=color.new(gapSColor, 70), text_size=size.small)
    row += 1

    // --- Filtres ---
    string candleStatus = candleSizeOk ? "OK" : "TROP GRANDE"
    color candleColor = candleSizeOk ? color.green : color.red
    table.cell(statusTable, 0, row, "Bougie", text_color=color.white, bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 1, row, na(currentCandlePct) ? "—" : str.tostring(currentCandlePct, "#.##") + "%", text_color=color.white, bgcolor=color.new(candleColor, 70), text_size=size.small)
    table.cell(statusTable, 2, row, "", bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 3, row, "", bgcolor=color.new(color.gray, 80), text_size=size.small)
    row += 1

    // Blackout
    string blackoutStatus = inBlackout ? "BLACKOUT" : "OK"
    color blackoutColor = inBlackout ? color.red : color.green
    table.cell(statusTable, 0, row, "Blackout", text_color=color.white, bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 1, row, blackoutStatus, text_color=color.white, bgcolor=color.new(blackoutColor, 70), text_size=size.small)
    table.cell(statusTable, 2, row, "", bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 3, row, "", bgcolor=color.new(color.gray, 80), text_size=size.small)
    row += 1

    // Position
    string posStatus = positionActive ? (currentSide == 1 ? "LONG" + (beTriggered ? " (BE)" : "") : "SHORT" + (beTriggered ? " (BE)" : "")) : "NONE"
    color posColor = positionActive ? (currentSide == 1 ? color.green : color.red) : color.gray
    table.cell(statusTable, 0, row, "Position", text_color=color.white, bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 1, row, posStatus, text_color=color.white, bgcolor=color.new(posColor, 70), text_size=size.small)
    table.cell(statusTable, 2, row, "", bgcolor=color.new(color.gray, 80), text_size=size.small)
    table.cell(statusTable, 3, row, "", bgcolor=color.new(color.gray, 80), text_size=size.small)
